# project is configured, the example is copied into the project handler

# project and it is never overwritten afterwards.

#========================== Handler module seeding ============================#

# Middlewares/ThirdParty/ModBus -> Middlewares
get_filename_component(MODBUS_THIRDPARTY_DIR "${CMAKE_CURRENT_LIST_DIR}" DIRECTORY)
get_filename_component(MODBUS_THIRDPARTY_DIR_NAME "${MODBUS_THIRDPARTY_DIR}" NAME)
get_filename_component(MODBUS_MIDDLEWARES_DIR "${MODBUS_THIRDPARTY_DIR}" DIRECTORY)

set(MODBUS_EXAMPLE_DIR  "${CMAKE_CURRENT_LIST_DIR}/Example")
set(MODBUS_HANDLER_DIR  "${MODBUS_MIDDLEWARES_DIR}/ModBus")

#------------------------------------------------------------------------------#
# Copies the example into project handler module folder Middlewares/ModBus if
# the example has not been copied yet (ModBus.cpp does not exist).
#
# The folder can already contain the empty module scaffold generated by EmBi
# platform (Setup - Configure middleware module, ModuleInit). The scaffold is
# replaced by the example only if it is still unmodified (scaffold ModBus.c
# with empty ModBus_Task), otherwise the folder is left untouched - it is the
# user's own handler module.
#
# HANDLER_SEEDED_ARG [out]: TRUE if the example was copied now
#------------------------------------------------------------------------------#
function(ModBus_SeedHandlerModule HANDLER_SEEDED_ARG)

    set(${HANDLER_SEEDED_ARG} FALSE PARENT_SCOPE)

    # Component is not vendored in a project (eg. standalone unit testing)
    if(NOT MODBUS_THIRDPARTY_DIR_NAME STREQUAL "ThirdParty")
        message(STATUS "ModBus: not located in Middlewares/ThirdParty, handler module is not created.")
        return()
    endif()

    if(EXISTS "${MODBUS_HANDLER_DIR}/ModBus.cpp")
        message(STATUS "ModBus: handler module ${MODBUS_HANDLER_DIR} found.")
        return()
    endif()

    set(SCAFFOLD_SOURCE "${MODBUS_HANDLER_DIR}/ModBus.c")

    if(EXISTS "${SCAFFOLD_SOURCE}")

        file(READ "${SCAFFOLD_SOURCE}" SCAFFOLD_CONTENT)

        # Unmodified scaffold of EmBi ModuleInit - empty task function
        if(NOT SCAFFOLD_CONTENT MATCHES "This function shall be called in the main loop of the application" OR
           NOT SCAFFOLD_CONTENT MATCHES "void ModBus_Task\\( void \\)[\r\n]+\\{[ \t\r\n]*\\}")
            message(WARNING "ModBus: ${MODBUS_HANDLER_DIR} contains user specific handler module (ModBus.c), "
                            "example is not copied. Delete the folder to get the example.")
            return()
        endif()

    else()

        file(GLOB HANDLER_FILES "${MODBUS_HANDLER_DIR}/*")

        if(HANDLER_FILES)
            message(WARNING "ModBus: ${MODBUS_HANDLER_DIR} is not empty, example is not copied. "
                            "Delete the folder to get the example.")
            return()
        endif()

    endif()

    file(MAKE_DIRECTORY "${MODBUS_HANDLER_DIR}")

    file(GLOB EXAMPLE_FILES LIST_DIRECTORIES false "${MODBUS_EXAMPLE_DIR}/*")

    foreach(EXAMPLE_FILE IN LISTS EXAMPLE_FILES)
        get_filename_component(EXAMPLE_FILE_NAME "${EXAMPLE_FILE}" NAME)
        file(COPY_FILE "${EXAMPLE_FILE}" "${MODBUS_HANDLER_DIR}/${EXAMPLE_FILE_NAME}")
    endforeach()

    # Scaffold source is replaced by ModBus.cpp
    file(REMOVE "${SCAFFOLD_SOURCE}")

    message(STATUS "ModBus: handler module created from example in ${MODBUS_HANDLER_DIR} - adapt it to the application.")

    set(${HANDLER_SEEDED_ARG} TRUE PARENT_SCOPE)

endfunction()


ModBus_SeedHandlerModule(MODBUS_HANDLER_SEEDED)

#============================ Stack library ===================================#

add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/ModBus)

#===================== Handler module processing check ========================#
# Middlewares.cmake processes Middlewares/ModBus before ThirdParty/. If the
# scaffold has been processed in this configuration, it was just replaced -
# the targets created from it are no longer valid.

if(MODBUS_HANDLER_SEEDED)

    if(TARGET ModBus_Lib)
        message(FATAL_ERROR "ModBus: handler module Middlewares/ModBus has been created from example "
                            "in this configuration (scaffold replaced). Configure the project again.")
    elseif(EXISTS "${MODBUS_HANDLER_DIR}/CMakeLists.txt")
        # Handler folder was not processed by Middlewares.cmake (did not exist yet)
        add_subdirectory("${MODBUS_HANDLER_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/Handler")
    endif()

endif()
